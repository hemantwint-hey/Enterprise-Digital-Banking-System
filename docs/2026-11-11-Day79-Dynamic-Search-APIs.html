Security Subtlety — LIKE Wildcard Injection (ReDoS / Full Scan): Criteria cb.like(...) prevents SQL injection via bind variables, but if a user sends ?branch=% or ?firstName=_____%, SQL treats % and _ as wildcards and forces full table scans! We escape \, %, and _ cleanly before wrapping in %...%.

package org.example.enterprisedigitalbankingsystem.common.util;

import jakarta.persistence.criteria.CriteriaBuilder;
import jakarta.persistence.criteria.Expression;
import jakarta.persistence.criteria.Predicate;

public final class SpecificationHelper {

    private static final char ESCAPE_CHAR = '\\';

    private SpecificationHelper() {}

    /** Escapes SQL LIKE wildcards ('%', '_', '\') and performs case-insensitive contains match. */
    public static Predicate containsIgnoreCase(
            CriteriaBuilder cb, Expression<String> field, String rawInput) {
        String escaped = rawInput.trim().toLowerCase()
                .replace("\\", "\\\\")
                .replace("%", "\\%")
                .replace("_", "\\_");
        return cb.like(cb.lower(field), "%" + escaped + "%", ESCAPE_CHAR);
    }

    public static boolean hasText(String value) {
        return value != null && !value.isBlank();
    }
}

@Data @NoArgsConstructor @AllArgsConstructor @Builder
public class TransactionSearchRequest {
    @Positive(message = "Account id must be positive") private Long accountId;
    @Size(max = 64) private String transactionReference;
    private TransactionType transactionType;
    private TransactionStatus transactionStatus;
    @DecimalMin("0.01") private BigDecimal minAmount;
    @DecimalMin("0.01") private BigDecimal maxAmount;
    @DateTimeFormat(iso = DateTimeFormat.ISO.DATE_TIME) private LocalDateTime fromDate;
    @DateTimeFormat(iso = DateTimeFormat.ISO.DATE_TIME) private LocalDateTime toDate;
    @Size(max = 100) private String remarksKeyword;

    @AssertTrue(message = "minAmount cannot exceed maxAmount")
    public boolean isAmountRangeValid() {
        return minAmount == null || maxAmount == null || minAmount.compareTo(maxAmount) <= 0;
    }

    @AssertTrue(message = "fromDate cannot be after toDate")
    public boolean isDateRangeValid() {
        return fromDate == null || toDate == null || !fromDate.isAfter(toDate);
    }
}

4. Planned Code: Spring Data 4.1 Specification Builders

Notice we use a clean List<Predicate> inside the lambda (avoiding deprecated Specification.where(null) from older Spring Data 2/3 codebases).

src/main/java/.../transaction/specification/TransactionSpecification.java NEW 
package org.example.enterprisedigitalbankingsystem.transaction.specification;

import jakarta.persistence.criteria.Predicate;
import org.example.enterprisedigitalbankingsystem.common.util.SpecificationHelper;
import org.example.enterprisedigitalbankingsystem.transaction.dto.request.TransactionSearchRequest;
import org.example.enterprisedigitalbankingsystem.transaction.entity.Transaction;
import org.springframework.data.jpa.domain.Specification;
import java.util.ArrayList;
import java.util.List;

public final class TransactionSpecification {
    private TransactionSpecification() {}

    public static Specification<Transaction> withFilters(TransactionSearchRequest filter) {
        return (root, query, cb) -> {
            List<Predicate> predicates = new ArrayList<>();
            if (filter == null) return cb.conjunction();

            if (filter.getAccountId() != null) {
                Predicate isSource = cb.equal(root.get("sourceAccount").get("id"), filter.getAccountId());
                Predicate isDest = cb.equal(root.get("destinationAccount").get("id"), filter.getAccountId());
                predicates.add(cb.or(isSource, isDest));
            }
            if (SpecificationHelper.hasText(filter.getTransactionReference())) {
                predicates.add(cb.equal(root.get("transactionReference"),
                        filter.getTransactionReference().trim()));
            }
            if (filter.getTransactionType() != null) {
                predicates.add(cb.equal(root.get("transactionType"), filter.getTransactionType()));
            }
            if (filter.getTransactionStatus() != null) {
                predicates.add(cb.equal(root.get("transactionStatus"), filter.getTransactionStatus()));
            }
            if (filter.getMinAmount() != null) {
                predicates.add(cb.greaterThanOrEqualTo(root.get("amount"), filter.getMinAmount()));
            }
            if (filter.getMaxAmount() != null) {
                predicates.add(cb.lessThanOrEqualTo(root.get("amount"), filter.getMaxAmount()));
            }
            if (filter.getFromDate() != null) {
                predicates.add(cb.greaterThanOrEqualTo(root.get("createdAt"), filter.getFromDate()));
            }
            if (filter.getToDate() != null) {
                predicates.add(cb.lessThanOrEqualTo(root.get("createdAt"), filter.getToDate()));
            }
            if (SpecificationHelper.hasText(filter.getRemarksKeyword())) {
                predicates.add(SpecificationHelper.containsIgnoreCase(
                        cb, root.get("remarks"), filter.getRemarksKeyword()));
            }
            return cb.and(predicates.toArray(new Predicate[0]));
        };
    }
}

public static Specification<Account> withFilters(AccountSearchRequest f) {
    return (root, query, cb) -> {
        List<Predicate> p = new ArrayList<>();
        if (f == null) return cb.conjunction();
        if (f.getCustomerId() != null)     p.add(cb.equal(root.get("customer").get("id"), f.getCustomerId()));
        if (hasText(f.getAccountNumber())) p.add(cb.equal(root.get("accountNumber"), f.getAccountNumber().trim()));
        if (f.getAccountType() != null)    p.add(cb.equal(root.get("accountType"), f.getAccountType()));
        if (f.getAccountStatus() != null)  p.add(cb.equal(root.get("accountStatus"), f.getAccountStatus()));
        if (hasText(f.getBranch()))        p.add(containsIgnoreCase(cb, root.get("branch"), f.getBranch()));
        if (f.getMinBalance() != null)     p.add(cb.greaterThanOrEqualTo(root.get("balance"), f.getMinBalance()));
        if (f.getMaxBalance() != null)     p.add(cb.lessThanOrEqualTo(root.get("balance"), f.getMaxBalance()));
        return cb.and(p.toArray(new Predicate[0]));
    };
}
