3. Relational Schema (DDL)

The DDL below is the schema Hibernate generates from the entities. It is the intended content of the Flyway baseline migration V1__baseline.sql PLANNED.

CREATE TABLE users (
  user_id         BINARY(16)   NOT NULL PRIMARY KEY,
  full_name       VARCHAR(255) NOT NULL,
  username        VARCHAR(255) NOT NULL UNIQUE,
  email           VARCHAR(255) NOT NULL UNIQUE,
  password        VARCHAR(255) NOT NULL,              -- BCrypt hash
  phone_number    VARCHAR(255) NOT NULL UNIQUE,
  role            VARCHAR(255) NOT NULL,              -- CUSTOMER|MANAGER|EMPLOYEE|ADMIN|AUDITOR
  status          VARCHAR(255) NOT NULL,              -- ACTIVE|INACTIVE|BLOCKED|SUSPENDED
  kyc_verified    BIT          NOT NULL,
  email_verified  BIT          NOT NULL,
  phone_verified  BIT          NOT NULL,
  last_login      DATETIME(6),
  created_at      DATETIME(6)  NOT NULL,
  updated_at      DATETIME(6)
);

CREATE TABLE customers (
  id              BIGINT AUTO_INCREMENT PRIMARY KEY,
  user_id         BINARY(16)   NOT NULL UNIQUE,
  first_name      VARCHAR(255) NOT NULL,
  last_name       VARCHAR(255) NOT NULL,
  date_of_birth   DATE,
  phone_number    VARCHAR(15)  NOT NULL UNIQUE,
  address         VARCHAR(255) NOT NULL,
  city            VARCHAR(255) NOT NULL,
  state           VARCHAR(255) NOT NULL,
  country         VARCHAR(255) NOT NULL,
  postal_code     VARCHAR(255) NOT NULL,
  kyc_status      VARCHAR(255) NOT NULL,              -- PENDING|IN_PROGRESS|VERIFIED|REJECTED
  pan_number      VARCHAR(10)  UNIQUE,
  aadhaar_number  VARCHAR(12)  UNIQUE,
  created_at      DATETIME(6)  NOT NULL,
  CONSTRAINT fk_customer_user FOREIGN KEY (user_id) REFERENCES users(user_id)
);

CREATE TABLE accounts (
  id                   BIGINT AUTO_INCREMENT PRIMARY KEY,
  account_number       VARCHAR(255)   NOT NULL UNIQUE,  -- "10" + 10 digits
  customer_id          BIGINT         NOT NULL,
  account_holder_name  VARCHAR(255)   NOT NULL,
  branch               VARCHAR(255)   NOT NULL,
  balance              DECIMAL(18,2)  NOT NULL,
  account_type         VARCHAR(255)   NOT NULL,       -- SAVINGS|CURRENT
  account_status       VARCHAR(255)   NOT NULL,       -- ACTIVE|BLOCKED|FROZEN|CLOSED
  created_at           DATETIME(6)    NOT NULL,
  updated_at           DATETIME(6),
  version              BIGINT,                        -- optimistic lock
  CONSTRAINT fk_account_customer FOREIGN KEY (customer_id) REFERENCES customers(id)
);

CREATE TABLE transactions (
  id                         BIGINT AUTO_INCREMENT PRIMARY KEY,
  transaction_reference      VARCHAR(255)  NOT NULL UNIQUE,  -- "TXN" + 13 digits
  source_account_id          BIGINT,
  destination_account_id     BIGINT,
  amount                     DECIMAL(18,2) NOT NULL,
  transaction_type           VARCHAR(255)  NOT NULL,   -- DEPOSIT|WITHDRAWAL|TRANSFER
  transaction_status         VARCHAR(255)  NOT NULL,   -- PENDING|SUCCESS|FAILED|REVERSED
  balance_after_transaction  DECIMAL(18,2) NOT NULL,
  remarks                    VARCHAR(255),
  created_at                 DATETIME(6)   NOT NULL,
  CONSTRAINT fk_txn_source FOREIGN KEY (source_account_id)      REFERENCES accounts(id),
  CONSTRAINT fk_txn_dest   FOREIGN KEY (destination_account_id) REFERENCES accounts(id)
);

CREATE TABLE ledger_entries (
  id                   BIGINT AUTO_INCREMENT PRIMARY KEY,
  transaction_id       BIGINT        NOT NULL,
  account_id           BIGINT        NOT NULL,
  entry_type           VARCHAR(255)  NOT NULL,         -- DEBIT|CREDIT
  amount               DECIMAL(18,2) NOT NULL,
  balance_after_entry  DECIMAL(18,2) NOT NULL,
  narration            VARCHAR(255),
  created_at           DATETIME(6)   NOT NULL,
  CONSTRAINT fk_ledger_txn     FOREIGN KEY (transaction_id) REFERENCES transactions(id),
  CONSTRAINT fk_ledger_account FOREIGN KEY (account_id)     REFERENCES accounts(id)
);

CREATE TABLE beneficiaries (
  id                      BIGINT AUTO_INCREMENT PRIMARY KEY,
  customer_id             BIGINT       NOT NULL,
  beneficiary_account_id  BIGINT       NOT NULL,
  nick_name               VARCHAR(255) NOT NULL,
  status                  VARCHAR(255) NOT NULL,       -- PENDING|ACTIVE|BLOCKED
  created_at              DATETIME(6)  NOT NULL,
  CONSTRAINT fk_benef_customer FOREIGN KEY (customer_id)            REFERENCES customers(id),
  CONSTRAINT fk_benef_account  FOREIGN KEY (beneficiary_account_id) REFERENCES accounts(id)
);

CREATE TABLE loans (
  id                       BIGINT AUTO_INCREMENT PRIMARY KEY,
  customer_id              BIGINT        NOT NULL,
  disbursement_account_id  BIGINT        NOT NULL,
  loan_type                VARCHAR(255)  NOT NULL,     -- PERSONAL|HOME|VEHICLE|EDUCATION
  principal_amount         DECIMAL(18,2) NOT NULL,
  interest_rate            DECIMAL(5,2)  NOT NULL,     -- annual %, e.g. 10.50
  tenure_months            INT           NOT NULL,
  emi_amount               DECIMAL(18,2) NOT NULL,
  status                   VARCHAR(255)  NOT NULL,     -- PENDING|ACTIVE|REJECTED|CLOSED
  created_at               DATETIME(6)   NOT NULL,
  approved_at              DATETIME(6),
  closed_at                DATETIME(6),
  CONSTRAINT fk_loan_customer FOREIGN KEY (customer_id)             REFERENCES customers(id),
  CONSTRAINT fk_loan_account  FOREIGN KEY (disbursement_account_id) REFERENCES accounts(id)
);

CREATE TABLE emis (
  id                  BIGINT AUTO_INCREMENT PRIMARY KEY,
  loan_id             BIGINT        NOT NULL,
  installment_number  INT           NOT NULL,
  due_date            DATE          NOT NULL,
  amount              DECIMAL(18,2) NOT NULL,
  status              VARCHAR(255)  NOT NULL,          -- PENDING|PAID|OVERDUE
  paid_at             DATETIME(6),
  CONSTRAINT fk_emi_loan FOREIGN KEY (loan_id) REFERENCES loans(id)
);

CREATE TABLE audit_logs (
  id            BIGINT AUTO_INCREMENT PRIMARY KEY,
  performed_by  BINARY(16),
  action        VARCHAR(255) NOT NULL,   -- CREATE|UPDATE|DELETE|LOGIN|LOGOUT|LOGIN_FAILED
  entity_name   VARCHAR(255) NOT NULL,
  entity_id     VARCHAR(255),
  old_value     LONGTEXT,
  new_value     LONGTEXT,
  description   VARCHAR(255),
  ip_address    VARCHAR(64),
  created_at    DATETIME(6)  NOT NULL,
  CONSTRAINT fk_audit_user FOREIGN KEY (performed_by) REFERENCES users(user_id)
);

5. Constraints and Integrity Rules

5.1 Database-enforced

TypeConstraint
Primary keysAll tables (UUID for users, identity BIGINT elsewhere, VARCHAR(36) for notification)
Uniqueusers.username, users.email, users.phone_number, customers.user_id, customers.phone_number, customers.pan_number, customers.aadhaar_number, accounts.account_number, transactions.transaction_reference, notification.idempotency_key
Foreign keysAll 13 relationships in section 2.1 (no ON DELETE CASCADE; deleting a parent with children fails)
NOT NULLAs per DDL in section 3
Optimistic lockaccounts.version incremented on every update

5.2 Application-enforced (service layer)

  • Balance never goes negative: withdraw/transfer fail with insufficient balance.
  • Only ACTIVE accounts can be source or destination of a transaction.
  • A transfer cannot have the same source and destination; exactly one of destinationAccountId / beneficiaryId must be given.
  • Transfers via beneficiary require status ACTIVE and the beneficiary must belong to the source account’s customer.
  • An account must have zero balance to be closed; a CLOSED account cannot be modified.
  • A customer cannot add the same account as a beneficiary twice.
  • A loan can be approved or rejected only while PENDING; an EMI cannot be paid twice.
  • PAN / Aadhaar / phone may not be linked to another customer.
  • Double-entry invariant: for every TRANSFER, Σ DEBIT = Σ CREDIT (verified by AccountingLogicTest).
