@ExtendWith(MockitoExtension.class)
class LoanServiceImplTest {
    @Mock LoanRepository loanRepository;      @Mock CustomerRepository customerRepository;
    @Mock AccountRepository accountRepository; @Mock EmiService emiService;
    @Spy  LoanMapper loanMapper = new LoanMapper();
    @InjectMocks LoanServiceImpl loanService;

    @Test
    void approveLoan_whenPending_disbursesAndGeneratesSchedule() {
        Account acc = Account.builder().id(1L).accountNumber("1000000001").balance(new BigDecimal("500.00")).build();
        Loan loan = Loan.builder().id(9L).customer(Customer.builder().id(1L).build()).disbursementAccount(acc)
                .principalAmount(new BigDecimal("10000.00")).status(LoanStatus.PENDING).build();
        when(loanRepository.findById(9L)).thenReturn(Optional.of(loan));
        when(loanRepository.save(any())).thenAnswer(i -> i.getArgument(0));

        LoanResponse res = loanService.approveLoan(9L);

        assertThat(res.getStatus()).isEqualTo(LoanStatus.ACTIVE);
        assertThat(acc.getBalance()).isEqualByComparingTo("10500.00");
        verify(emiService).generateSchedule(loan);
    }
}

8.2 k6 script (sketch)

import http from 'k6/http';
import { check } from 'k6';
export const options = { vus: 50, duration: '2m', thresholds: { http_req_duration: ['p(95)<250'] } };
const H = { headers: { 'Content-Type': 'application/json', Authorization: `Bearer ${__ENV.TOKEN}` } };
export default function () {
  const src = 1 + Math.floor(Math.random() * 50), dst = 51 + Math.floor(Math.random() * 50);
  const r = http.post('http://localhost:8080/api/v1/transactions/transfer',
      JSON.stringify({ sourceAccountId: src, destinationAccountId: dst, amount: 1 }), H);
  check(r, { 'status 200': (x) => x.status === 200 });
}
